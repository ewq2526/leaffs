# Camphor 引导菜单主题
#
# 属性只用手册第 7 章列出的那些（Theme file format）：
#   https://www.gnu.org/software/grub/manual/grub/html_node/Theme-file-format.html
# 手册里也写明：注释行只有在上一行不期待取值时才算注释，所以组件块里不写注释。
#
# 同一份 theme.txt 两处用：介质上的 /boot/grub/live-theme/、装好系统里的
# /boot/grub/themes/camphor/。底图与标题图都在同目录，引用写相对路径。
#
# boot_menu 的框要按"条目都放得下"给：grub 不会把框撑开，超出框高的条目直接卷到
# 看不见（边上只有一根滚动条），而屏幕右边大半空着。以前框高 24% 只有四行多一点，
# 条目一多就成了"只看得到四条、剩下的不知道去哪了"；宽度 16% 也放不下长标题。

desktop-image: "splash.jpg"
desktop-image-scale-method: "crop"
title-text: ""

+ image {
	file = "title.png"
	left = 6%
	top = 6%
}

+ boot_menu {
	left = 6%
	top = 18%
	width = 46%
	height = 72%
	item_font = "Unifont Regular 16"
	item_color = "#c6d2d8"
	selected_item_color = "#a6e6bb"
	item_height = 36
	item_padding = 4
	item_spacing = 16
	item_icon_space = 0
	icon_width = 0
	icon_height = 0
	item_pixmap_style = "item_*.png"
	selected_item_pixmap_style = "select_*.png"
}

+ label {
	top = 100%-34
	left = 6%
	width = 40%
	height = 20
	text = "@KEYMAP_SHORT@"
	align = "left"
	color = "#8b98a2"
	font = "Unifont Regular 16"
}
