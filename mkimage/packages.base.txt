# 主安装镜像的包清单：基本系统。
#
# 这一层只到「内核 + systemd 骨架 + 能装系统的最小工具」。
# 界面、文件服务、共享、无线这些不在这里 —— 它们在应用及服务镜像里，
# 由安装完成后的脚本从那个镜像铺进来。
#
# 一行一个包名；# 之后是理由，不参与解析。
# 与 mkimage/packages.app.txt、mkimage/packages.installer.txt 三份合起来才是整套系统。

# ── 内核与固件 ──────────────────────────────────────────────
# 内核按具体版本写：构建用本地包仓库，里面只有这一份 deb，没有 meta 包。
# 换内核时同时改这里与仓库里的 deb。
# 模块树是裁过的（850 个，33 MB）：网卡蓝牙整棵保留，声音、媒体、InfiniBand 不要。
linux-image-6.12.113+deb13-amd64
initramfs-tools            # 生成 initrd，配置用仓库的 /etc/initramfs-tools
zstd                       # 那份配置里 COMPRESS=zstd，重做 initrd 时要用
firmware-realtek
firmware-iwlwifi
firmware-atheros
firmware-brcm80211
firmware-intel-graphics    # i915
firmware-misc-nonfree
firmware-mediatek          # MT7921/7922 与 mt76x0 的无线、蓝牙固件（模块树里都有）
firmware-ath9k-htc         # Atheros AR9271 USB 无线
firmware-carl9170          # Atheros AR9170 USB 无线
firmware-intel-misc        # Intel 蓝牙（ibt-*.sfi）与 e100 网卡
wireless-regdb
intel-microcode
amd64-microcode            # 与 intel-microcode 对称：AMD 平台的 CPU 微码

# ── 图形：X 与窗口 ──────────────────────────────────────────
# 安装器和应用都要用，所以放在主镜像这一层，两个支镜像共用的就是这一份。
xserver-xorg-core
xserver-xorg-input-libinput
xserver-xorg-legacy        # xinit 由 root 起 X 时需要
xinit
x11-xserver-utils
openbox                    # 会话里的窗口管理器，只承载一个全屏窗口
libgl1-mesa-dri            # i915 的 GL 实现
librsvg2-common            # GTK 的 SVG 图标加载器
# 界面中文的字体必须在主镜像这一层：安装器跑在 live 里，应用层要到首次启动才铺进来，
# 装系统那一步一个中文字都没有。字体也不是"应用的东西"，两个支镜像共用它。
# 用 Droid Sans Fallback（7 MB，一个 ttf）：Debian 稳定版里的老包，之前那一版镜像
# 里就是它。Noto CJK 那套 91 MB，简繁日韩、明体黑体全在，用不到。
fonts-droid-fallback
# 界面里用 ✅ ❌ ⚙ ⚠ 这些符号表示状态，DejaVu 与 Droid 都没有前两个的字形，
# 缺了就是方框。Noto Color Emoji 是 Debian 里的标准 emoji 字体，11 MB。
fonts-noto-color-emoji

# ── 窗口里的那个视图 ────────────────────────────────────────
gir1.2-webkit2-4.1         # 安装器与应用都靠它显示页面
gir1.2-freedesktop         # Gdk 依赖的 cairo/xlib typelib，缺了 gi.repository.Gdk 导入失败
python3-gi
python3-gi-cairo
publicsuffix               # libpsl 的域名后缀表，WebKit 判 cookie 域要用

# ── 预览页的播放 ────────────────────────────────────────────
# 预览页把文件挂到 <video src="/download/…"> 上播（preview.js），WebKitGTK 的解码
# 走 GStreamer：这一组不在，视频只有声音没有画面，或者干脆不动。
gstreamer1.0-libav
gstreamer1.0-plugins-bad
gstreamer1.0-gl
gstreamer1.0-x

# ── Python ─────────────────────────────────────────────────
# 安装器的后端与应用都是 Python 写的，同在公共层。
python3
python3-venv

# ── 存储 ───────────────────────────────────────────────────
e2fsprogs
dosfstools                 # ESP 与 U 盘
rsync                      # 应用镜像铺进系统时用它

# ── 网络 ───────────────────────────────────────────────────
# systemd-networkd 负责拿地址（DHCP），但它不写 /etc/resolv.conf —— 那是
# systemd-resolved 的活。以前没装 resolved，DHCP 给的 DNS 无处落地，
# /etc/resolv.conf 是 0 字节：外网域名全部解析不了，timesyncd 连 NTP 服务器
# 都找不到（时钟因此一直不同步）。
systemd-resolved
# 时钟：机器没有可靠的 RTC，界面走 HTTPS、证书按时间校验。以前这个包是被应用层
# 的 Recommends 顺带拉进来的，落在首次启动才装 —— 第一次开机根本不同步。写进这
# 一层，主镜像就有它，构建时按清单启用，第一次开机就跑。
systemd-timesyncd

# ── 基础 ───────────────────────────────────────────────────
locales
console-setup
keyboard-configuration
tzdata
dbus                       # logind 与各服务的会话总线
linux-sysctl-defaults      # 内核默认参数，不自带会改变一批 sysctl 默认值
ca-certificates
sudo

# ── 引导 ───────────────────────────────────────────────────
# Secure Boot 那条链：装到盘上的 grub 必须是签名版，并由微软签名的 shim 拉起来。
grub-efi-amd64-signed
shim-signed
efibootmgr
# 装好的系统里 update-grub 要靠 os-prober 把同盘上的其它系统写进菜单。
os-prober
