# Camphor 系统包清单
#
# 镜像的包来源只有这一份文件：从 Debian 13 的 minbase 装起，再装下面这些。
# 不从已安装好的系统复制 —— 那样带进来的是那套环境的历史，既不可复现，
# 也把发布版撑大好几倍。
#
# 用法：由 mkimage/BUILD.md 第 1 步转成 live-build 的包清单
#   sed 's/#.*//' packages.txt | grep -v '^$' > config/package-lists/camphor.list.chroot
#
# 一行一个包名；# 之后是理由，不参与解析。
# 不列入：编译器、qemu、git、桌面环境、语音合成与无障碍组件。

# ── 内核与固件 ──────────────────────────────────────────────
# 内核按普通包装：auto/config 里的 --linux-packages none 只关掉 live-build
# 自己装卸内核那一步，内核由下面这一行从包源装，不从运行中的系统拷文件。
# 写具体版本而不是 linux-image-amd64：构建用的是本地包仓库，
# 里面只有这一份内核 deb，没有 meta 包。换内核时同步改这里与仓库里的 deb。
linux-image-6.12.113+deb13-amd64
initramfs-tools            # 生成 initrd，配置用仓库的 /etc/initramfs-tools
firmware-realtek           # Realtek 无线网卡
firmware-iwlwifi
firmware-atheros
firmware-brcm80211
firmware-intel-graphics    # i915
firmware-misc-nonfree
wireless-regdb
intel-microcode

# ── 图形：X 与窗口 ──────────────────────────────────────────
xserver-xorg-core
xserver-xorg-input-libinput
xserver-xorg-legacy        # xinit 由 root 起 X 时需要
xinit
x11-xserver-utils
openbox                    # 会话里的窗口管理器，只承载 kiosk 那一个窗口
libgl1-mesa-dri            # i915 的 GL 实现

# ── 窗口里的那个视图 ────────────────────────────────────────
gir1.2-webkit2-4.1         # pywebview 的 GTK 后端
python3-gi
python3-gi-cairo

# ── Python ─────────────────────────────────────────────────
python3

aria2                      # 下载器
ffmpeg                     # 视频预览
fonts-noto-cjk             # 界面中文

# ── 网络、共享与发现 ────────────────────────────────────────
openssh-server
ksmbd-tools
avahi-daemon
wsdd2
libnss-mdns
wpasupplicant
iw
rfkill
nftables                   # modules-load.d/camphor-netfilter.conf
systemd-timesyncd

# ── 引导 ───────────────────────────────────────────────────
# os-prober：装好的系统里 update-grub 要靠它把同盘上的其它系统（例如 Windows）
# 找出来写进菜单。安装器那条路也一样，收尾时会重跑一次 update-grub。
# 配合 /etc/default/grub 里的 GRUB_DISABLE_OS_PROBER=false —— 新版 grub 默认关闭它。
os-prober

dosfstools                 # ESP 与 U 盘
rsync

locales
console-setup
keyboard-configuration
tzdata

ca-certificates
